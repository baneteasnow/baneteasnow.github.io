<script>
	const YEAR_BOOKS_API_URL =
		"https://script.google.com/macros/s/AKfycbyfpo2_zPkA2rjL2ZB_8y4RaH_3g7poEOM-oUNhgUXfRqUujklW8_8uTxZXBFGS5AF6/exec?mode=books&year=2026"

    const READING_YEAR = 2026
    const READING_GOAL = Number("{{ page.goal }}") || 70

    function escapeHTML(value) {
    	return String(value ?? "")
    		.replaceAll("&", "&amp;")
    		.replaceAll("<", "&lt;")
    		.replaceAll(">", "&gt;")
    		.replaceAll('"', "&quot;")
    		.replaceAll("'", "&#039;")
    }

    async function loadYearBooks() {
    	const tableBody = document.getElementById("yearBooksBody")
    	const header = document.getElementById("readingGoalHeader")
    	const errorEl = document.getElementById("yearBooksError")

    	try {
    		const response = await fetch(YEAR_BOOKS_API_URL, {
    			cache: "no-store",
    		})

    		if (!response.ok) {
    			throw new Error(`HTTP ${response.status}`)
    		}

    		const data = await response.json()
    		const books = data.books || []

    		const readCount = books.length
    		const percentage =
    			READING_GOAL > 0
    				? Math.round((readCount / READING_GOAL) * 100)
    				: 0

    		header.textContent =
    			`${readCount} / ${READING_GOAL} ${percentage}%`

    		tableBody.innerHTML = books
    			.map((book) => {
    				const month =
    					escapeHTML(book.month) +
    					" " +
    					String(READING_YEAR).slice(-2)

    				const title = book.year
    					? `${escapeHTML(book.title)} (${escapeHTML(book.year)})`
    					: escapeHTML(book.title)

    				return `
    					<tr>
    						<td style="text-align: center">
    							${escapeHTML(book.id)}
    						</td>

    						<td>
    							${title}
    						</td>

    						<td>
    							${escapeHTML(book.author)}
    						</td>

    						<td>
    							${escapeHTML(book.pages)}
    						</td>

    						<td>
    							${escapeHTML(book.format)}
    						</td>

    						<td>
    							${month}
    						</td>
    					</tr>
    				`
    			})
    			.join("")

    		updateReadingProgress(readCount)
    	} catch (error) {
    		console.error(error)

    		tableBody.innerHTML = ""
    		errorEl.style.display = "block"
    	}
    }

    function updateReadingProgress(readCount) {
    	const progress = document.getElementById("readingChallengeProgress")
    	const progressText = document.getElementById("readingChallengeText")

    	if (progress) {
    		progress.value = readCount
    		progress.max = READING_GOAL
    		progress.title = `${readCount}/${READING_GOAL}`
    	}

    	if (progressText) {
    		progressText.textContent = `${readCount}/${READING_GOAL}`
    	}
    }

    loadYearBooks()

</script>

<script>
	const YEAR_STATS_API_URL =
	"https://script.google.com/macros/s/AKfycbyfpo2_zPkA2rjL2ZB_8y4RaH_3g7poEOM-oUNhgUXfRqUujklW8_8uTxZXBFGS5AF6/exec?year=2026"

    const months = [
    	"Jan",
    	"Feb",
    	"Mar",
    	"Apr",
    	"May",
    	"Jun",
    	"Jul",
    	"Aug",
    	"Sep",
    	"Oct",
    	"Nov",
    	"Dec",
    ]

    async function loadYearStats() {
    	const booksRow = document.getElementById("booksRow")
    	const pagesRow = document.getElementById("pagesRow")
    	const errorEl = document.getElementById("yearStatsError")

    	try {
    		const response = await fetch(YEAR_STATS_API_URL, {
    			cache: "no-store",
    		})

    		if (!response.ok) {
    			throw new Error(`HTTP ${response.status}`)
    		}

    		const data = await response.json()

    		booksRow.innerHTML =
    			months
    				.map((month) => `<td>${data.months[month].books}</td>`)
    				.join("") +
    			`<td>${data.total.books}</td>` +
    			`<td>books</td>`

    		pagesRow.innerHTML =
    			months
    				.map((month) => `<td>${data.months[month].pages}</td>`)
    				.join("") +
    			`<td>${data.total.pages}</td>` +
    			`<td>pages</td>`
    	} catch (error) {
    		console.error(error)

    		booksRow.innerHTML = ""
    		pagesRow.innerHTML = ""
    		errorEl.style.display = "block"
    	}
    }

    loadYearStats()

</script>
